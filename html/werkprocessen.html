Opdracht 2: Realiseren (K1-W3)

import { Modal, Pressable, Text, View } from "react-native";
import * as Constants from "@/src/Styles/constants";
import { styles } from "@/src/Styles/styles";
import { CategoryItem, TaskItem } from "@/src/Types/types";

type confirmationModalProps = {
    item: TaskItem | CategoryItem | null,
    onCancel: () => void,
    onConfirm: (id: TaskItem | CategoryItem) => void
}

export default function ConfirmationModal({
    item,
    onCancel,
    onConfirm,
}: confirmationModalProps) {
    return (
        <Modal visible={!!item} transparent animationType='fade'>
            <View style={styles.confirmationModalWrapper}>
                <View style={styles.confirmationModalContainer}>
                    <Text style={styles.confirmationModalTitle}>
                        Delete "<Text style={{{{ color: Constants.tertriaryColor }}}}>
                            {item?.title}
                        </Text>"?
                    </Text>
                    <View style={{
                        flexDirection: 'row',
                        justifyContent: 'space-around',
                        marginTop: 'auto',
                        marginBottom: 24,
                    }}>
                        <Pressable onPress={onCancel}>
                            <Text style={styles.confirmationModalButton}>
                                Never mind
                            </Text>
                        </Pressable>
                        <Pressable onPress={() => item && onConfirm(item)}>
                            <Text style={styles.confirmationModalDelete}>
                                Delete
                            </Text>
                        </Pressable>
                    </View>
                </View>
            </View>
        </Modal>
    )
}

Bovenstaande code is een voorbeeld van een React Native component genaamd "ConfirmationModal", die een modaal venster weergeeft voor het bevestigen van een delete-actie binnen de TaskManager-applicatie. De component maakt gebruik van TypeScript voor typeveiligheid en bevat props voor het item dat verwijderd moet worden, evenals callback-functies voor het annuleren of bevestigen van de actie.
